## install using pip3 ##
########################

Use pip3 install -U for latest versions

1) install miniconda

2) Go to (base) environment
   conda activate

3) Create conda environment:
   conda create -n data python=3.12 -y

4) Activate the environment:
   conda activate data

5) Install libraries:
   pip3 install -U termcolor regex loguru pytz numpy pandas datar[pandas] polars matplotlib seaborn plotnine plotly scipy statsmodels sympy cvxpy scikit-learn imblearn xgboost hdbscan mlxtend pyECLAT umap-learn[plot] pmdarima

#######################
## Remove old things ##
#######################

# Remove all ROCm packages
sudo apt autoremove rocm-core rocm-libs rocm-dev amd-smi -y
sudo apt purge rocm-* amdgpu-* -y
sudo apt autoremove -y

# Remove AMDGPU install script
sudo apt purge amdgpu-install -y

# Remove repositories
sudo rm -f /etc/apt/sources.list.d/amdgpu*.list
sudo rm -f /etc/apt/sources.list.d/rocm*.list

# Clean cache
sudo rm -rf /var/cache/apt/*
sudo apt clean all
sudo apt update

############################
## Install Ryzen software ##
############################

# Install amdgpu-install (7.1.0)
sudo apt update
wget https://repo.radeon.com/amdgpu-install/7.2/ubuntu/noble/amdgpu-install_7.2.70200-1_all.deb
sudo apt install ./amdgpu-install_7.2.70200-1_all.deb
sudo apt update

# Install kernel headers first
sudo apt install "linux-headers-$(uname -r)"

# Install ROCm stack (without DKMS if you want to keep existing kernel driver)
sudo amdgpu-install --usecase=rocm --no-dkms -y

# Add user
sudo usermod -a -G render,video $LOGNAME

# Reboot to apply changes
sudo reboot

# Verify ROCM installation
rocminfo
amd-smi
